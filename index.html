### EXTRACT ACTIVE TRAVEL PARTS ###

timeToNextPoint5min <- function(traj_x) {
  # make sure ordered by time
  traj_x <- traj_x[order(traj_x$datetime), ]
  n <- nrow(traj_x)
  # output vector
  time_to_next_5min <- rep(NA_real_, n)
  for(i in 1:n) {
    current_time <- traj_x$datetime[i]
    # time differences in seconds
    diffs <- as.numeric(
      difftime(
        traj_x$datetime[(i+1):n],
        current_time,
        units = "secs"
      )
    )
    # first point at least 5 min (=300 sec) ahead
    idx <- which(diffs >= 300)[1]
    if(!is.na(idx)) {
      time_to_next_5min[i] <- diffs[idx]
    }
  }
  # add column
  traj_x$timeToNextPoint5min_sec <- time_to_next_5min
  return(traj_x)
}

distToNextPoint5min <- function(traj_x) {
  # make sure ordered by time
  traj_x <- traj_x[order(traj_x$datetime), ]
  n <- nrow(traj_x)
  # output vector
  time_to_next_5min <- rep(NA_real_, n)
  for(i in 1:n) {
    current_time <- traj_x$datetime[i]
    # time differences in seconds
    diffs <- as.numeric(
      difftime(
        traj_x$datetime[(i+1):n],
        current_time,
        units = "secs"
      )
    )
    # first point at least 5 min (=300 sec) ahead
    idx <- which(diffs >= 300)[1]
    if(!is.na(idx)) {
      time_to_next_5min[i] <- diffs[idx]
    }
  }
  # add column
  traj_x$timeToNextPoint5min_sec <- time_to_next_5min
  return(traj_x)
}

distToNextPoint5min <- function(traj_x) {
  # ensure ordering
  traj_x <- traj_x[order(traj_x$datetime), ]
  n <- nrow(traj_x)
  # output vector (meters)
  dist_to_next_5min <- rep(NA_real_, n)
  # convert to sf points (WGS84)
  sf_pts <- st_as_sf(traj_x, coords = c("lon", "lat"), crs = 4326)
  # loop over points
  for(i in 1:n) {
    current_time <- traj_x$datetime[i]
    # find future points
    future_idx <- which(traj_x$datetime > current_time)
    if(length(future_idx) == 0) next
    # time diffs (seconds)
    dt <- as.numeric(difftime(traj_x$datetime[future_idx],
                              current_time,
                              units = "secs"))
    # first index >= 5 minutes
    j <- which(dt >= 300)[1]
    if(!is.na(j)) {
      target_idx <- future_idx[j]
      # distance in meters using sf
      dist <- st_distance(sf_pts[i, ], sf_pts[target_idx, ], by_element = TRUE)
      dist_to_next_5min[i] <- as.numeric(dist)
    }
  }
  traj_x$distToNextPoint5min_m <- dist_to_next_5min
  return(traj_x)
}

speedToNextPoint5min <- function(traj_x){
  traj_x$speedToNextPoint5min_ms <- traj_x$distToNextPoint5min_m / traj_x$timeToNextPoint5min_sec
  traj_x$distToNextPoint5min_m[is.na(traj_x$speedToNextPoint5min_ms)] <- NA
  return(traj_x)
}

# speed threashold: 0.833 m/s (=3km/h)
travelling <- function(traj_x) {
  traj_x$travelling <- traj_x$speedToNextPoint5min_ms >= 0.833
  traj_x$travelling[is.na(traj_x$speedToNextPoint5min_ms)] <- NA
  
  return(traj_x)
}

extractMovingSegment <- function(traj_x) {
  # find all TRUE positions
  moving_idx <- which(traj_x$travelling == TRUE)
  # if no movement exists
  if(length(moving_idx) == 0) {
    return(NULL)
  }
  # first moving point
  start_idx <- moving_idx[1]
  # find first FALSE after movement started
  after_start <- traj_x$travelling[start_idx:nrow(traj_x)]
  end_false <- which(after_start == FALSE)[1]
  # determine end index
  if(is.na(end_false)) {
    end_idx <- nrow(traj_x)
  } else {
    end_idx <- start_idx + end_false - 2
  }
  # return only consecutive moving segment
  traj_moving <- traj_x[start_idx:end_idx, ]
  return(traj_moving)
}

for(nm in traj_names) {
  
  cat("Processing", nm, "...\n")
  
  # load trajectory
  traj <- get(nm)
  
  # processing pipeline
  traj <- timeToNextPoint5min(traj)
  traj <- distToNextPoint5min(traj)
  traj <- speedToNextPoint5min(traj)
  traj <- travelling(traj)
  
  # extract moving segment
  traj_moving <- extractMovingSegment(traj)
  
  # save objects in workspace
  assign(nm, traj)
  assign(paste0(nm, "_moving"), traj_moving)
  
  # export to csv
  write.csv(
    traj_moving,
    file = file.path(
      "Data/Preprocessed",
      paste0(nm, "_moving.csv")
    ),
    row.names = FALSE
  )
}

add_step_distance <- function(traj) {
  traj <- traj %>%
    arrange(datetime)
  # shift geometry to align each point with the next one
  geom_next <- lead(st_geometry(traj))
  # compute point-to-next-point distance (meters)
  traj$step_dist_m <- as.numeric(
    st_distance(traj, geom_next, by_element = TRUE)
  )
  # last point has no next point
  traj$step_dist_m[is.na(traj$step_dist_m)] <- 0
  traj
}
 
# dist to next point
for (nm in names_traj) {
  traj <- get(nm)
  traj <- add_step_distance(traj)
  assign(nm, traj)
}

# L0: direct path from A to B
add_total_distance <- function(traj) {
  traj <- traj %>%
    arrange(datetime)
  first_pt <- traj[1, ]
  last_pt  <- traj[nrow(traj), ]
  total_dist <- as.numeric(
    st_distance(first_pt, last_pt)
  )
  traj$direct_dist <- total_dist
  traj
}

for (nm in names_traj) {
  traj <- get(nm)
  traj <- add_total_distance(traj)
  assign(nm, traj)
}

# L1: actual path from A to B
add_total_path_distance <- function(traj) {
  traj <- traj %>%
    arrange(datetime)
  traj$total_path_dist_m <- sum(traj$step_dist_m, na.rm = TRUE)
  traj
}

for (nm in names_traj) {
  traj <- get(nm)
  traj <- add_total_path_distance(traj)
  assign(nm, traj)
}

# L1/L0: directness of actual path
for (nm in names_traj) {
  traj <- get(nm)
  traj$path_directness <- traj$total_path_dist_m / traj$direct_dist
  assign(nm, traj)
}

Abstract

Human mobility is constrained by transportation networks and physical barriers, leading to travel paths that deviate from the straight-line distance between origins and destinations. This deviation can be quantified using trajectory directness, defined as the ratio between travelled distance and Euclidean distance. This study investigates whether longer journeys exhibit higher directness using a subsample of 79 trajectories from the GeoLife GPS Trajectories Dataset, which contains long-term GPS records from 182 users. For each trajectory, total path length and straight-line distance were computed using spatial operations in R, and a directness ratio was derived. The relationship between journey length and directness was evaluated using Spearman rank correlation. Results show a weak but statistically significant negative correlation (ρ = -0.297, p = 0.008), indicating that longer journeys tend to be more direct. However, substantial variability is observed, particularly in short trajectories. This pattern is likely influenced by infrastructure effects and preprocessing decisions, including trajectory segmentation at pauses longer than five minutes, which may differentially affect transport modes. Overall, the findings suggest that journey length is associated with increased route efficiency, but that measurement and mode-related biases play an important role in observed trajectory structure.

Introduction

Human mobility is a central aspect of everyday life. Although travelers often seek efficient routes between origins and destinations, real-world trajectories rarely follow the shortest possible path. Instead, movement is constrained by transportation networks, topography, and other physical barriers, causing travelled distances to exceed the straight-line distance between start and end points.

The difference between travelled distance and straight-line distance can be used to quantify the directness of a trajectory. Directness provides a simple measure of how efficiently a journey connects its origin and destination.

This study investigates the directness of real-world travel trajectories using the GeoLife GPS Trajectories Dataset, a large collection of GPS tracks recorded by 182 users over more than five years. For each trajectory, travelled distance and straight-line distance are compared to calculate a directness ratio. The relationship between journey length and directness is then examined. It is hypothesized that longer journeys exhibit higher directness than shorter journeys.

Material and Methods

The analysis was based on trajectory data from the GeoLife GPS Trajectories Dataset (Zheng et al. 2009). To reduce computational complexity while retaining a diverse set of movement patterns, a random subsample of approximately 100 trajectories was selected for analysis.

Trajectory Preprocessing. Only active travel segments were included in the study. Movement was classified as active when the recorded speed exceeded 3 km/h over the following five consecutive minutes. Trajectory sequences were required to contain a minimum of two GPS observations to allow the calculation of movement metrics.

Measures of Trajectory Directness. To quantify the directness of movement, three measures were calculated for each trajectory using the R package sf (Pebesma, 2023):

  1. Total trajectory length – the cumulative distance travelled along the recorded trajectory.

  2. Direct length – the straight-line (Euclidean) distance between the trajectory’s start and end points.

  3. Directness ratio – calculated as the ratio between direct length and total trajectory length:

Directness Ratio = Total Trajectory Length / Direct Length

Values close to 1 indicate highly direct movement, whereas lower values indicate more circuitous trajectories. Trajectory directness was used as an indicator of movement efficiency, describing how closely observed movement paths approximated the shortest possible route between origin and destination.

Statistical Analysis. The relationship between trajectory length and directness was investigated using Spearman’s rank correlation coefficient. Spearman’s correlation was chosen because it does not assume normally distributed variables and is robust to non-linear monotonic relationships.

Visualisation. Data exploration and result presentation were conducted in R. Scatterplots were created using ggplot2 (Wickham, 2016) to visualize relationships between trajectory characteristics and directness measures. Spatial representations of trajectories were produced using tmap (Tennekes, 2024), allowing visual assessment of movement patterns and route geometry.

The analysis was done in R (R Core Team, 2025). ChatGPT (OpenAI, 2026) was used as a coding assistant.

However, the distribution was strongly right-skewed, with most trajectories being relatively short (< 5 km) and only a few trajectories exceeding several tens of kilometers. Discussion

The results support the hypothesis that longer journeys are more direct, with a weak but significant negative correlation between direct distance and trajectory directness (ρ = -0.297, p = 0.008). However, the relationship is highly scattered, indicating that journey length alone is a limited predictor of route efficiency.

A likely explanation is that longer trips provide access to more efficient, infrastructure-driven routes (e.g., highways or rail networks), increasing optimization opportunities over larger spatial scales. In contrast, short-distance movement is more affected by local constraints and detours, leading to higher variability in directness.

Several methodological factors may also bias the results. Trajectories were split at pauses longer than five minutes, which may fragment trips containing layovers (e.g., train transfers or EV charging) and remove intermediate detours. This likely inflates directness estimates and reduces observed path length.

This bias is not uniform across transport modes: public transport and electric vehicles are more likely to include long stops, whereas walking, cycling, and conventional car travel are less affected. As a result, apparent differences in directness may partly reflect segmentation artefacts rather than true movement efficiency.

Overall, while the hypothesis is supported, trajectory directness is strongly influenced by data preprocessing and transport-mode composition, limiting direct interpretation of efficiency alone.
